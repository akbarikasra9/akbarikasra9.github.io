    col.lab = dark_purple,

    bty = "l",

# Remove missing HPI scores
scores <- na.omit(hpi_2025$HPI)

# Group countries by HPI score
hpi_groups <- cut(
  scores,
  breaks = c(-Inf, 30, 40, 50, Inf),
  labels = c(
    "Below 30",
    "30-39",
    "40-49",
    "50 or above"
  ),
  right = FALSE
)

# Count countries in each category
group_counts <- table(hpi_groups)

# Remove categories with zero countries
group_counts <- group_counts[group_counts > 0]

# Calculate percentages
percentages <- round(
  100 * group_counts / sum(group_counts),
  1
)

# Custom Viridis colors

  deep_blue,
  teal,
  green
)

  group_counts,

  ) +

  # Chart labels

    title = "Top 5 Countries by Happy Planet Index (2025)",

  ) +

  # Apply custom theme from histogram
  theme_mine() +

Histogram

Base R was easier to produce the histogram because hist() generated the chart with little codes. However, ggplot2 was easier for customizing the mean reference line and keeping a consistent visual style.

Vertical Bar Chart

For this chart, ggplot2 was easier because I could assign various colors to each nation, adjust the the nation labels, and add values above the bars. Base R required more processes to position the labels and draw the gridlines.

Horizontal Bar Chart

I preferred ggplot2 because it was easier use to arrange nations, match their colors with the vertical bar chart, and position the HPI scores. Base R in this case required more manual adjustments to get the same appearance.

Pie Chart

Base R was easier to utilize because the pie() function created the chart directly from the nation counts. In this instance, ggplot2 required more code to create the slices, calculate label positions, and add the connector lines.

Boxplot

I found ggplot2 easier because I couldn assign different colors to each HPI group and add individual country observations using geom_jitter(). Base R required separate functions for the boxplot and individual observations.

Scatterplot

I preferred ggplot2 because geom_smooth() made it easier to add the regression line without manually fitting and drawing the model. Additionally, it was easier to customize the country dots, gridlines, and legend while maintaining the same theme as my other charts

# Reusable Base R boxplot function
draw_base_boxplot <- function() {

  op <- par(no.readonly = TRUE)

  my_style()

  # Adjust margins for 6.5 x 4 inch output
  par(mar = c(7, 5, 3, 1.5))

  # Create boxplot
  boxplot(
    `Life Expectancy` ~ HPI_Group,
    data = box_data,
    col = box_colors,
    border = dark_purple,
    main = "Life Expectancy by HPI Score Range (2025)",
    xlab = "",
    ylab = "Life expectancy (years)",
    las = 1,
    lwd = 1.5,
    outline = FALSE,
    cex.main = 0.9,
    cex.lab = 0.85,
    cex.axis = 0.8
  )

  # Add individual country observations
  set.seed(6356)

  stripchart(
    `Life Expectancy` ~ HPI_Group,
    data = box_data,
    vertical = TRUE,
    method = "jitter",
    jitter = 0.12,
    add = TRUE,
    pch = 19,
    cex = 0.45,
    col = adjustcolor(deep_blue, alpha.f = 0.35)
  )

  # X-axis title
  mtext(
    "Happy Planet Index score range (points)",
    side = 1,
    line = 3,
    cex = 0.85,
    col = dark_purple
  )

  # Legend in bottom-left margin
  legend(
    x = grconvertX(0.02, from = "ndc", to = "user"),
    y = grconvertY(0.12, from = "ndc", to = "user"),
    legend = levels(box_data$HPI_Group),
    fill = box_colors,
    border = NA,
    bty = "n",
    cex = 0.65,
    xjust = 0,
    yjust = 0,
    xpd = NA
  )

  # Source note
  mtext(
    source_note,
    side = 1,
    line = 5.5,
    cex = 0.6,
    col = deep_blue
  )

  par(op)
}

3.e Recommended Output Formats

Journal Article — PDF: I would choose PDF because it preserves the quality of the text, dots, and boxplot lines when enlarged. Based on my comparison at 400% zoom, the PDF maintained sharp details without noticeable pixelation, making it suitable for academic publications.

Web Page — SVG: I would choose SVG because it maintains sharp text, dots, and lines when zoomed in. When I compared both SVG boxplots at 400%, neither showed noticeable pixelation, making SVG a good choice for websites viewed on different screen sizes.

PowerPoint Slide — PNG: I would choose PNG because it is widely supported in PowerPoint and uses lossless compression. Although some pixelation can occur when enlarged, exporting at 300 DPI provides sufficient resolution for most presentations.

4. Reflection

Even though Murrell includes pie charts in R Graphics, he acknowledges their limitations. Cleveland and McGill (1984) state that people judge positions and lengths more accurately than angles and areas, making bar charts better for precise comparisons. I believe, on the other hand, that pie charts are acceptable under some conditions.

A pie chart should contain only a few categories, between three to five, so the slices are easy to differentiate. Furthermore, there should be clear variations between categories, particularly when one or two account for a large proportion. Aditionally, the categories must demonstrate portions of a complete whole, totaling 100%. Lastly, pie charts are better suited for showing proportions at one point in time rather than changes over time.

My Happy Planet Index pie charts are examples of such. They are made up of three categories representing 100% of nations with available HPI scores in 2025. Countries scoring 50 or above account for 46.3%, followed by 44% scoring 40–49 and 9.7% scoring 30–39. The charts show the differences in the smallest category, although a bar chart would make comparing the two largest categories simpler.
