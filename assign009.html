Part C. Reactive Texas Change Atlas Dashboard

For Part C, I developed an interactive Shiny dashboard that extends my Assignment 8 Texas Change Atlas.

The dashboard includes four sections: Where, Who, How fast, and Map. The new Map section uses Leaflet to display county-level changes in earnings, educational attainment, and unemployment across Texas.

Users can select indicators, filter county classifications, compare measurement periods, and explore individual counties through interactive maps. The dashboard also includes summary statistics, Dallas–Fort Worth and Houston detail maps, and a companion chart.

# All county CHANGE views use one semantic palette. Source-provided status
# controls direction. Intensity is split within each indicator at the median
# absolute difference among reliably classified closing/widening counties.
# These bins are descriptive; they are NOT additional significance tests.
change_colors <- c(
  "Widening · larger" = "#A73B24",
  "Widening · smaller" = "#EF9B84",
  "No clear change" = "#D3D4D1",
  "Closing · smaller" = "#70CDB4",
  "Closing · larger" = "#0C6B58",
  "Use with caution" = "#E5E3DE",
  "No estimate" = "#F1F2F4"
)
change_main <- change_colors[c("Widening · larger", "Widening · smaller",
                               "No clear change", "Closing · smaller",
                               "Closing · larger")]
county <- county |>
  mutate(change_magnitude = if_else(is.finite(g19) & is.finite(g24),
                                    abs(g24 - g19), NA_real_))
change_thresholds <- county |>
  filter(status %in% c("Closing", "Widening"), is.finite(change_magnitude)) |>
  group_by(metric) |>
  summarise(threshold = median(change_magnitude), .groups = "drop")
get_threshold <- function(m) {
  x <- change_thresholds$threshold[change_thresholds$metric == m]
  if (!length(x) || !is.finite(x[1])) return(0)
  unname(x[1])
}
classify_change <- function(status, g19, g24, metric) {
  status <- as.character(status)
  mag <- abs(g24 - g19)
  threshold <- get_threshold(metric)
  result <- rep("No estimate", length(status))
  paired <- is.finite(g19) & is.finite(g24)
  result[paired & status == "No clear change"] <- "No clear change"
  result[paired & status == "Use with caution"] <- "Use with caution"
  result[paired & status == "Closing" & mag <= threshold] <- "Closing · smaller"
  result[paired & status == "Closing" & mag > threshold] <- "Closing · larger"
  result[paired & status == "Widening" & mag <= threshold] <- "Widening · smaller"
  result[paired & status == "Widening" & mag > threshold] <- "Widening · larger"
  result
}
unit_of_change <- function(m) if (m == "earnings") "index points" else "percentage points"
# Signed ranking coordinate from the source's direction + absolute observed
# movement. Positive = source classified as Closing; negative = Widening.
# This is NOT the algebraic delta and is only assigned to reliable classes.
signed_change <- function(cls, magnitude) {
  ifelse(grepl("^Closing", cls), magnitude,
         ifelse(grepl("^Widening", cls), -magnitude, NA_real_))
}
class_names <- c("Widening · larger" = "Widened more",

                 "Closing · larger" = "Closed more",

# All counties appear in the count legend (filtered county count when a
# county-type filter is used); caution is a separate keyed symbol.
legend_html <- function(d, metric) {
  counts <- table(factor(d$change_class, levels = names(change_colors)))
  keys <- names(change_main)
  tile <- vapply(keys, function(k) {
    sprintf('<span class="legend-cell"><span class="legend-swatch" style="background:%s"></span><span>%s <b>(%d)</b></span></span>',
            change_colors[[k]], class_names[[k]], as.integer(counts[[k]]))
  }, character(1))
  note <- sprintf('Smaller/larger observed shifts are separated at %.1f %s for this indicator; not a new significance test.',
                  get_threshold(metric), unit_of_change(metric))
  HTML(paste0('<div class="map-legend-row">', paste(tile, collapse = ''),
              '</div><div class="map-legend-extra">',
              '<b>×</b> Use with caution (', counts[["Use with caution"]], ') &nbsp; ',
              '<span class="legend-swatch" style="background:#F1F2F4"></span> ',
              'No paired estimate (', counts[["No estimate"]], ')',
              '</div><div class="map-legend-note">', note, '</div>'))
}

# County bubble plots: color = SOURCE change category, symbol = county type,
# size = population. No numerical conclusions are implied by symbol shape.
county_bubble <- function(m) {
  d <- county |>
    filter(metric == m, !is.na(g24), !is.na(pct_fb),
           status != "No estimate") |>
    mutate(point_size = 8 + 24 * sqrt(pmax(pop, 0) /
                           max(pmax(pop, 0), na.rm = TRUE)),
           metro_symbol = case_when(
             is_large_metro(metro) ~ "circle",
             grepl("rural|nonmetro", tolower(coalesce(metro, ""))) ~ "diamond",
             TRUE ~ "square"
           ),
           hover = paste0("<b>", county_name, "</b><br>County type: ", metro,
                          "<br>", p2, " value: ", fmt1(g24),
                          "<br>", p1, " value: ", fmt1(g19),
                          "<br>Reported change: ", status,
                          "<br>Foreign-born residents: ", fmt1(pct_fb), "%"))
  p <- plot_base(paste0("Where are county differences largest? · ", metric_titles[[m]]),
                 "Each point = county · X = foreign-born share · Y = 2020–2024 value · bubble size = population")
  for (st in intersect(names(status_colors), unique(d$status))) {
    z <- d |> filter(status == st)
    p <- p |> add_trace(
      data = z, x = ~pct_fb, y = ~g24, type = "scatter", mode = "markers",
      name = st, text = ~hover, hoverinfo = "text",
      marker = list(color = status_colors[[st]], size = z$point_size,
                    symbol = z$metro_symbol, opacity = 0.88,
                    line = list(color = "#233443", width = 0.8))
    )
  }
  p |> layout(
    xaxis = list(title = "Foreign-born residents (% of county population)"),
    yaxis = list(title = metric_titles[[m]]),
    legend = list(orientation = "h", y = -0.26, x = 0.05),
    margin = list(t = 95, b = 115, l = 85, r = 20)
  )
}

# The Where-tab static map follows exactly the same change categories as
# the new interactive map. Gray means No clear change, not a zero gap.
county_status_map <- function() {
  m <- map_sf |>
    filter(metric == "earnings") |>
    mutate(change_class = factor(
      classify_change(status, g19, g24, "earnings"),
      levels = names(change_colors))) |>
    st_transform(5070)
  count_levels <- names(change_colors)
  counts <- table(factor(m$change_class, levels = count_levels))
  legend_labels <- paste0(unname(class_names[count_levels]), " (", as.integer(counts), ")")
  closing_n <- sum(counts[grepl("^Closing", count_levels)])
  widening_n <- sum(counts[grepl("^Widening", count_levels)])
  caution_n <- counts[["Use with caution"]]
  caution <- m |> filter(change_class == "Use with caution") |> st_point_on_surface()
  metros <- m |> filter(is_large_metro(metro))
  ggplot() +
    geom_sf(data = m, aes(fill = change_class), color = "#8193A2", linewidth = 0.20) +
    geom_sf(data = metros, fill = NA, color = "#263445", linewidth = 0.48) +
    geom_sf_text(data = caution, aes(label = "×"), color = "#4A5563", size = 2.7) +
    scale_fill_manual(
      values = change_colors,
      breaks = count_levels,
      labels = legend_labels,
      drop = FALSE,
      name = NULL
    ) +
    labs(
      title = paste0("Earnings-gap change across Texas counties"),
      subtitle = paste0(closing_n, " narrowed · ", widening_n, " widened · ", caution_n, " caution"),
      caption = NULL
    ) +
    coord_sf(datum = NA, expand = TRUE) +
    theme_void(base_size = 12) +
    theme(
      plot.title = element_text(size = 16, face = "bold", hjust = 0.5),
      plot.subtitle = element_text(size = 11, color = "#536274", hjust = 0.5),
      plot.caption = element_text(size = 8.8, color = "#536274", hjust = 0),
      legend.position = "left",
      legend.direction = "vertical",
      legend.justification = "center",
      legend.background = element_rect(fill = "white", color = "#D7DEE4"),
      legend.box.margin = margin(0, 12, 0, 0),
      legend.key.height = grid::unit(0.30, "in"),
      legend.key.width = grid::unit(0.35, "in"),
      legend.text = element_text(size = 9),
      plot.margin = margin(15, 16, 10, 16)
    ) +
    guides(fill = guide_legend(ncol = 1, byrow = TRUE))

    t,

    options = list(

# Comparison charts: two observation windows, not continuous annual trends.
# Display the time windows clearly and define each index in plain language.
generation_scatter <- function() {
  d <- gen_e |>
    mutate(
      period_label = if_else(period == "2015-2019", "2015–2019", "2022–2024"),

      education = if_else(deg, "College degree", "No college degree"),
      sex = if_else(female, "Women", "Men"),
      group = paste(generation, education, sex, sep = " · ")
    )
  
  ggplot(d, aes(x = period_label, y = score, group = group, color = generation)) +
    geom_hline(yintercept = 100, linetype = "dashed", color = "#7c8795", linewidth = 0.5) +

    geom_point(size = 2.8) +
    facet_grid(education ~ sex) +
    scale_color_manual(values = who_colors) +
    labs(
      title = "Earnings by generation, education, and sex",
      subtitle = "Two survey periods compared; dashed line marks equal earnings (100)",
      x = "Survey period (not individual years)",
      y = "Earnings relative to comparison group (100 = equal pay)",
      color = "Generation",
      caption = "Source: Texas Change Atlas / IPUMS; descriptive group comparisons"
    ) +
    theme_minimal(base_size = 12) +
    theme(
      plot.title = element_text(face = "bold", size = 14),
      plot.subtitle = element_text(size = 10, color = "#526173"),
      axis.text.x = element_text(size = 10),
      panel.grid.minor = element_blank(),

      strip.text = element_text(face = "bold")
    )
}

generation_gaps <- function() {

    mutate(

        str_detect(str_to_lower(as.character(measure)), "unemploy") ~ "Unemployment-rate gap",

      caption = "Source: Texas Change Atlas / IPUMS; sign interpretation follows source definitions"
    ) +
    theme_minimal(base_size = 12) +

# Separate cohort panels make age-of-arrival patterns easier to follow.
# The score is an index, NOT annual salary or a rate of earnings growth.
arrival_lines <- function() {
  # Cohort colors are distinct but coordinated with the Where palette.
  # They identify arrival cohorts only and do NOT encode better/worse outcomes.
  cohort_palette <- cohort_palette_match
  a <- arrival |>
    arrange(cohort_start, ysm) |>
    mutate(cohort = factor(as.character(cohort),
                           levels = unique(as.character(cohort))))
  cohort_levels <- levels(a$cohort)
  if (length(cohort_levels) > length(cohort_palette)) {
    stop("More than eight arrival cohorts: expand cohort_palette before plotting")
  }
  cohort_colors <- setNames(cohort_palette[seq_along(cohort_levels)], cohort_levels)

  ggplot(a, aes(x = ysm, y = score, group = cohort, color = cohort)) +
    geom_hline(yintercept = 100, linetype = "dashed", color = "#7c8795") +
    geom_line(linewidth = 1.1, na.rm = TRUE) +
    geom_point(size = 2.7, na.rm = TRUE) +
    facet_wrap(~cohort, ncol = 4, scales = "free_x",
               labeller = labeller(cohort = function(x) paste("Arrived", x))) +
    scale_color_manual(values = cohort_colors, guide = "none") +
    scale_x_continuous(breaks = scales::breaks_pretty(n = 4)) +
    labs(
      title = "Earnings comparisons by time since arrival",
      subtitle = "Each panel is a different arrival cohort; points are survey-group comparisons, not individual earnings histories",
      x = "Years lived in the U.S. at the time of the survey",
      y = "Earnings index (100 = equal pay)",
      caption = "Dashed line: earnings parity (100). Below 100 means lower relative earnings.\nSource: Texas Change Atlas / IPUMS; descriptive, not causal."
    ) +
    theme_minimal(base_size = 12) +
    theme(
      plot.title = element_text(face = "bold", size = 17),
      plot.subtitle = element_text(size = 11, color = "#526173"),
      strip.text = element_text(face = "bold", size = 11),
      panel.grid.minor = element_blank(),
      plot.caption = element_text(color = "#536274", hjust = 0),
      plot.margin = margin(14, 18, 14, 18)
    )
}

server <- function(input, output, session) {
  # Assignment 8: existing chart views
  output$earnings_bubble <- renderPlotly(county_bubble("earnings"))
  output$degree_bubble <- renderPlotly(county_bubble("degree"))
  output$unemployment_bubble <- renderPlotly(county_bubble("unemployment"))
  output$status_map <- renderPlot(county_status_map(), res = 110)
  output$county_table <- renderDT(county_table(), server=FALSE)
  output$generation_earnings <- renderPlot(generation_scatter(), res = 110)
  output$generation_gaps <- renderPlot(generation_gaps(), res = 110)
  output$arrival_lines <- renderPlot(arrival_lines(), res = 110)
  
  
  
  # Full county layer for the selected indicator. This preserves all 254 county
  # shapes as background context even when the county-type filter is narrower.
  map_all <- reactive({
    req(input$metric, input$period)
    d <- map_sf |> filter(metric == input$metric)
    d$value <- as.numeric(d[[input$period]])
    d$display_name <- coalesce(as.character(d$county_name), d$GEOID)
    d$change_class <- classify_change(d$status, d$g19, d$g24, input$metric)
    d$change_signed <- signed_change(d$change_class, abs(d$g24 - d$g19))
    d
  })

  # One joined sf reactive drives the three maps, statistics, legend, and bar chart.
  # Source statuses are not recomputed.
  dat <- reactive({
    req(input$metric, input$metro, input$period)
    d <- map_all()
    if (!identical(input$metro, "All counties")) {
      d <- d |> filter(!is.na(metro), metro == input$metro)
    }
    d
  })
  current_unit <- reactive(metric_titles[[input$metric]])
  output$total_counties <- renderText(nrow(dat()))
  output$n_counties <- renderText(sum(is.finite(dat()$value)))
  output$n_missing <- renderText(sum(!is.finite(dat()$value)))
  output$median_value <- renderText({
    x <- dat()$value
    x <- x[is.finite(x)]
    if (!length(x)) "No data" else sprintf("%.1f", median(x))
  })
  output$map_heading <- renderUI({
    d <- st_drop_geometry(dat())
    closing <- sum(grepl("^Closing", d$change_class))
    widening <- sum(grepl("^Widening", d$change_class))
    undecided <- sum(d$change_class == "No clear change")
    ttl <- paste0(closing, " counties narrowed the reported gap; ",
                  widening, " widened it")
    subt <- paste0(current_unit(), " · Change between ", p1, " and ", p2,
                   " · ", undecided, " counties have no clear change")
    div(class = "map-heading", tags$h3(ttl), tags$p(subt))
  })
  output$map_legend <- renderUI(legend_html(st_drop_geometry(dat()), input$metric))

  prepared_map <- reactive({
    d <- dat()
    unit <- unit_of_change(input$metric)
    raw_delta <- d$g24 - d$g19
    signed_fmt <- ifelse(is.finite(raw_delta), sprintf("%+.1f", raw_delta), "Not available")
    d$popup_text <- paste0(
      "<b>", htmltools::htmlEscape(d$display_name), "</b>",
      "<br><b>Indicator:</b> ", htmltools::htmlEscape(current_unit()),
      "<br>", p1, ": ", fmt_map(d$g19),
      "<br>", p2, ": ", fmt_map(d$g24),
      "<br><b>Arithmetic change:</b> ", signed_fmt, " ", unit,
      "<br><b>Reported change category:</b> ", htmltools::htmlEscape(d$change_class),
      "<br><b>County type:</b> ", htmltools::htmlEscape(coalesce(as.character(d$metro), "Unknown")),
      "<br>Foreign-born resident share: ",
      ifelse(is.finite(d$pct_fb), paste0(sprintf("%.1f", d$pct_fb), "%"), "Not available"),
      "<br><small>MOE and sample size: not included in this mapping file.</small>",
      "<br><small>Color direction follows the original source classification.</small>"
    )
    d
  })

  metro_featured <- c("Harris", "Dallas", "Tarrant", "Bexar",
                      "Travis", "El Paso", "Hidalgo", "Webb")
  # County text shown sparingly at statewide level. Inset maps show more names.
  map_layers <- function(map, d, mode = "state", with_legend = FALSE) {
    # Always construct polygons from the complete 254-county source layer.
    # The county-type filter changes emphasis and outputs, not the geometry.
    all_counties <- map_all()
    if (nrow(all_counties) == 0) return(map)
    selected_ids <- d$GEOID
    all_counties$selected <- all_counties$GEOID %in% selected_ids
    all_counties$shown_class <- ifelse(
      all_counties$selected, all_counties$change_class, "Not selected"
    )
    all_counties$fill_hex <- unname(change_colors[all_counties$change_class])
    all_counties$fill_hex[is.na(all_counties$fill_hex)] <- "#E4E8ED"
    all_counties$fill_hex[!all_counties$selected] <- "#E4E8ED"
    all_counties$popup_text <- paste0(
      "<b>", htmltools::htmlEscape(all_counties$display_name), "</b>",
      "<br>Change classification: ",
      htmltools::htmlEscape(unname(class_names[all_counties$change_class])),
      "<br>", p1, ": ", fmt_map(all_counties$g19),
      "<br>", p2, ": ", fmt_map(all_counties$g24),
      ifelse(all_counties$selected, "", "<br>Outside selected county type")
    )

    # One fully opaque polygon layer prevents the underlying OSM basemap from
    # washing out the pale counties. Borders stay visible even in rural Texas.
    map <- map |> addPolygons(
      data = all_counties,
      fillColor = all_counties$fill_hex,
      fillOpacity = 1,
      color = "#8595A4", weight = 0.85,
      smoothFactor = 0,
      layerId = ~GEOID,
      popup = all_counties$popup_text,
      label = paste0(all_counties$display_name, " — ",
                     unname(class_names[all_counties$change_class])),
      highlightOptions = highlightOptions(
        color = "#172B3D", weight = 2, bringToFront = TRUE
      )
    )

    # Eight featured county outlines are independent of the county filter.
    featured <- all_counties |> filter(is_featured(display_name))
    if (nrow(featured) > 0) {
      map <- map |> addPolygons(
        data = featured, fill = FALSE,
        color = "#273849", weight = 2,
        options = pathOptions(clickable = FALSE)
      )
    }

    cautions <- all_counties |> filter(
      selected, change_class == "Use with caution"
    )
    if (nrow(cautions) > 0) {
      pts <- inside_points(cautions)
      map <- map |> addLabelOnlyMarkers(
        data = pts, label = rep("×", nrow(pts)),
        labelOptions = labelOptions(
          noHide = TRUE, direction = "center", textOnly = TRUE, opacity = 1,
          style = list("color" = "#263445", "font-size" = "16px",
                       "font-weight" = "bold", "text-shadow" = "0px 0px 3px white")
        )
      )
    }

    if (isTRUE(input$show_labels)) {
      labels <- all_counties
      if (mode == "state") labels <- labels |> filter(is_featured(display_name))
      if (mode == "dfw") labels <- labels |> filter(grepl(
        "Dallas|Tarrant|Collin|Denton|Rockwall|Kaufman|Ellis|Johnson|Parker|Wise|Hood|Hunt",
        display_name))
      if (mode == "houston") labels <- labels |> filter(grepl(
        "Harris|Fort Bend|Brazoria|Galveston|Montgomery|Liberty|Chambers|Waller",
        display_name))
      if (nrow(labels) > 0) {
        label_points <- inside_points(labels)
        short_names <- gsub(" County$", "", label_points$display_name)
        map <- map |> addLabelOnlyMarkers(
          data = label_points, label = short_names,
          labelOptions = labelOptions(
            noHide = TRUE, direction = "center", textOnly = FALSE,
            opacity = 0.96,
            style = list("font-size" = if (mode == "state") "11px" else "10px",
                         "font-weight" = "bold", "color" = "#263445",
                         "background" = "rgba(255,255,255,0.89)",
                         "padding" = "2px 4px", "border-radius" = "3px")
          )
        )
      }
    }
    map
  }

  map_extent <- list(
    texas = c(-106.9, 25.6, -93.3, 36.8),
    dfw = c(-98.0, 32.1, -96.1, 33.6),
    houston = c(-96.3, 28.8, -94.4, 30.7)
  )
  make_initial_map <- function(mode) {
    d <- isolate(prepared_map())
    extent <- map_extent[[if (mode == "state") "texas" else mode]]
    leaflet(data = d, options = leafletOptions(zoomControl = TRUE, scrollWheelZoom = TRUE)) |>
      addTiles(options = tileOptions(maxZoom = 18)) |>
      fitBounds(extent[1], extent[2], extent[3], extent[4]) |>
      map_layers(d, mode)
  }
  output$texas_map <- renderLeaflet(make_initial_map("state"))
  output$dfw_map <- renderLeaflet(make_initial_map("dfw"))
  output$houston_map <- renderLeaflet(make_initial_map("houston"))

  # Proxy update, with all tiles remaining in place after filter changes.
  observeEvent(list(input$metric, input$metro, input$period, input$show_labels),
    ignoreInit = TRUE, {
      d <- prepared_map()
      for (id in c("texas_map", "dfw_map", "houston_map")) {
        mode <- switch(id, texas_map = "state", dfw_map = "dfw", houston_map = "houston")
        p <- leafletProxy(id, data = d, session = session) |>
          clearShapes() |> clearMarkers() |> clearControls()
        map_layers(p, d, mode)
      }
    }
  )
  observeEvent(input$reset_maps, {
    for (id in c("texas_map", "dfw_map", "houston_map")) {
      loc <- switch(id, texas_map = "texas", dfw_map = "dfw", houston_map = "houston")
      e <- map_extent[[loc]]
      leafletProxy(id, session = session) |>
        fitBounds(e[1], e[2], e[3], e[4])
    }
  })
  # Companion chart uses the same sf reactive as all maps and value boxes.
  output$top10 <- renderPlot({
    d <- dat() |> st_drop_geometry() |>
      filter(is.finite(change_signed))
    validate(need(nrow(d) > 0,
       "No counties with reliable closing/widening classifications for these filters."))
    closed <- d |> filter(change_signed > 0) |>
      slice_max(change_signed, n = 5, with_ties = FALSE)
    widened <- d |> filter(change_signed < 0) |>
      slice_min(change_signed, n = 5, with_ties = FALSE)
    ranked <- bind_rows(closed, widened) |> arrange(change_signed)
    ggplot(ranked, aes(x = change_signed,
                      y = forcats::fct_reorder(display_name, change_signed),
                      fill = ifelse(change_signed >= 0, "Closed", "Widened"))) +
      geom_vline(xintercept = 0, color = "#354052", linewidth = 0.7) +
      geom_col(width = 0.7) +
      geom_text(aes(label = sprintf("%+.1f", change_signed)),
                hjust = ifelse(ranked$change_signed < 0, 1.1, -0.1), size = 3.5) +
      scale_fill_manual(values = c("Closed" = "#0C6B58", "Widened" = "#D6654A"),
                        name = "Source change category") +
      scale_x_continuous(expand = expansion(mult = c(0.22, 0.22))) +
      labs(title = "Largest reported narrowing and widening shifts",
           subtitle = paste0("Source-reported direction; bar length = absolute value of the recorded change (", unit_of_change(input$metric), ")"),
           x = paste0("Widening  ←  Observed shift magnitude (", unit_of_change(input$metric), ")  →  Narrowing"),
           y = NULL,
           caption = "Only source-classified Closing/Widening counties; source classification sets direction. No independent 90% MOE tests performed.") +
      theme_minimal(base_size = 13) +
      theme(legend.position = "bottom", plot.title = element_text(face = "bold"),
            panel.grid.major.y = element_blank(),
            plot.caption = element_text(hjust = 0),
            plot.margin = margin(12, 28, 12, 12))
  })
